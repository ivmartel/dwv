/**
 * Minimal ViewController mock for annotation tests.
 * All methods return safe defaults; individual tests override as needed.
 */
class MockViewController {
  #uid;
  #modality;
  #acquisitionOrientation;
  #planeHelper;

  /**
   * @param {object} opts Configuration options.
   * @param {string} opts.uid Image SOP instance UID.
   * @param {string} [opts.modality] Modality string (default 'CT').
   * @param {boolean} [opts.acquisitionOrientation] Acquisition orientation
   *   flag.
   * @param {object} [opts.planeHelper] Plane helper mock.
   */
  constructor({
    uid = 'uid-test',
    modality = 'CT',
    acquisitionOrientation = true,
    planeHelper = {isAquisitionOrientation: () => true, getCosines: () => []}
  } = {}) {
    this.#uid = uid;
    this.#modality = modality;
    this.#acquisitionOrientation = acquisitionOrientation;
    this.#planeHelper = planeHelper;
  }

  /** @returns {{length: Function, get: Function}} Position-like object. */
  getCurrentPosition() {
    return {length: () => 3, get: () => undefined};
  }

  /** @returns {string} The SOP instance UID. */
  getCurrentImageUid() {
    return this.#uid;
  }

  /** @returns {string} SOP class UID. */
  getSopClassUid() {
    return '1.2.840.10008.5.1.4.1.1.2';
  }

  /**
   * @param {string} _uid The UID to look up.
   * @returns {Point3D} The plane origin.
   */
  getOriginForImageUid(_uid) {
    return new Point3D(0, 0, 0);
  }

  /** @returns {boolean} Whether in acquisition orientation. */
  isAquisitionOrientation() {
    return this.#acquisitionOrientation;
  }

  /**
   * @param {object} _position Current position.
   * @returns {Point3D[]} Plane points.
   */
  getPlanePoints(_position) {
    return [
      new Point3D(0, 0, 0),
      new Point3D(1, 0, 0),
      new Point3D(0, 1, 0)
    ];
  }

  /**
   * @param {string} uid UID to check.
   * @returns {boolean} True if this controller owns that UID.
   */
  includesImageUid(uid) {
    return uid === this.#uid;
  }

  /** @returns {object} The plane helper. */
  getPlaneHelper() {
    return this.#planeHelper;
  }

  /** @returns {string} The modality. */
  getModality() {
    return this.#modality;
  }
}

   */
  test('Annotation constructor sets unique non-empty ids and default values',
    () => {
      const a = new Annotation();

  /**
   * Tests for {@link Annotation} meta data add/get/list operations.
   *
   * @function module:tests/image~annotationMetaAddGet
   */
  test('Annotation addMetaItem / getMetaItem / getMetaConceptIds', () => {
    const ann = new Annotation();

    // empty initially
    assert.deepEqual(ann.getMetaConceptIds(), []);
    assert.isUndefined(ann.getMetaItem('C0001'));

    // add one item
    const concept = new DicomCode('Finding');
    concept.value = 'C0001';
    const value = new DicomCode('Lesion');
    ann.addMetaItem(concept, value);

    assert.deepEqual(ann.getMetaConceptIds(), ['C0001']);
    const item = ann.getMetaItem('C0001');
    assert.equal(item.concept, concept, 'stored concept');
    assert.equal(item.value, value, 'stored value');

    // add a second item
    const concept2 = new DicomCode('Quality');
    concept2.value = 'C0002';
    ann.addMetaItem(concept2, 'good');
    assert.deepEqual(
      ann.getMetaConceptIds().sort(), ['C0001', 'C0002']
    );
  });

  /**
   * Tests that {@link Annotation#addMetaItem} warns when overwriting.
   *
   * @function module:tests/image~annotationMetaOverwrite
   */
  test('Annotation addMetaItem warns on duplicate concept id', () => {
    const warnSpy = vi.spyOn(loggerModule.logger, 'warn')
      .mockImplementation(() => {});
    const ann = new Annotation();
    const concept = new DicomCode('Finding');
    concept.value = 'C0001';

    ann.addMetaItem(concept, 'first');
    assert.equal(warnSpy.mock.calls.length, 0, 'no warning on first add');

    ann.addMetaItem(concept, 'second');
    assert.equal(warnSpy.mock.calls.length, 1, 'warning on overwrite');
    assert.ok(warnSpy.mock.calls[0][0].includes('C0001'),
      'warning mentions the concept id');
  });

  /**
   * Tests for {@link Annotation#removeMetaItem}.
   *
   * @function module:tests/image~annotationMetaRemove
   */
  test('Annotation removeMetaItem deletes an existing entry silently', () => {
    const ann = new Annotation();
    const concept = new DicomCode('Finding');
    concept.value = 'C0001';
    ann.addMetaItem(concept, 'val');

    ann.removeMetaItem('C0001');
    assert.isUndefined(ann.getMetaItem('C0001'));
    assert.deepEqual(ann.getMetaConceptIds(), []);

    // removing a non-existent id is a no-op
    assert.doesNotThrow(() => ann.removeMetaItem('no-such-id'));
  });

  test('Annotation isCompatibleView delegates to planeHelper ' +

  /**
   * Tests for {@link Annotation#isCompatibleView} with planePoints set.
   *
   * @function module:tests/image~annotationCompatibleViewPlane
   */
  test('Annotation isCompatibleView compares cosines when planePoints set',
    () => {
      const ann = new Annotation();
      ann.planePoints = [
        new Point3D(0, 0, 0),
        new Point3D(1, 0, 0), // cosine row-direction
        new Point3D(0, 1, 0) // cosine column-direction
      ];

      const matchingHelper = {
        getCosines: () => [1, 0, 0, 0, 1, 0],
        getLegacyCosines: () => [1, 0, 0, 0, 1, 0]
      };
      assert.isTrue(ann.isCompatibleView(matchingHelper),
        'matches when cosines are equal');

      const nonMatchingHelper = {
        getCosines: () => [0, 1, 0, 0, 0, 1],
        getLegacyCosines: () => [0, 1, 0, 0, 0, 1]
      };
      assert.isFalse(ann.isCompatibleView(nonMatchingHelper),
        'does not match when cosines differ');

      const legacyHelper = {
        getCosines: () => [1, 0, 0, 0, -1, 0],
        getLegacyCosines: () => [1, 0, 0, 0, 1, 0]
      };
      assert.isTrue(ann.isCompatibleView(legacyHelper),
        'matches when legacy cosines are equal');
    }
  );

  test('Annotation init sets a 1-based referencedFrameNumber', () => {
    for (const timeIndex of [0, 2]) {
      const ann = new Annotation();
      const vc = new MockViewController({uid: 'sop-001'});
      // the position holds a volume id, the index the volume index
      vc.getCurrentPosition = () => ({

        get: (i) => (i === 3 ? timeIndex * 500 : 0)
      });
      vc.getIndexFromPosition = (position) => ({
        get: (i) => (i === 3 ? position.get(3) / 500 : 0)
      });
      ann.init(vc);
      assert.equal(ann.referencedFrameNumber, timeIndex + 1,
        `frame number for time index ${timeIndex}`);

  });

  /**
   * Tests that {@link Annotation#getCentroid} uses the volume id
   * of the referenced frame.
   *
   * @function module:tests/image~annotationCentroidVolumeId
   */
  test('Annotation getCentroid with volume ids', () => {
    // 2 slices, 3 volumes with ids 0, 500 and 1000
    const geometry = new Geometry(
      [new Point3D(0, 0, 0)], new Size([4, 4, 1]), new Spacing([1, 1, 1]),
      undefined, {id: {source: 'DiffusionBValue', value: 0}});
    geometry.appendOrigin(new Point3D(0, 0, 1), 1, 0);
    for (const value of [500, 1000]) {
      geometry.appendFrame(
        new Point3D(0, 0, 0), {id: {source: 'DiffusionBValue', value}});
      geometry.appendOrigin(new Point3D(0, 0, 1), 1, value);
    }

    let position = geometry.indexToWorld(new Index([0, 0, 0, 2]));
    const vc = new MockViewController({uid: 'sop-001'});
    vc.getCurrentPosition = () => position;
    vc.getCurrentIndex = () => geometry.worldToIndex(position);
    vc.getIndexFromPosition = (point) => geometry.worldToIndex(point);
    vc.getPositionFromIndex = (index) => geometry.indexToWorld(index);
    vc.getScrollDimIndex = () => 2;
    vc.getPositionFromPlanePoint = (point, k) =>
      new Point([point.getX(), point.getY(), k, position.get(3)]);

    const ann = new Annotation();
    ann.init(vc);
    assert.equal(ann.referencedFrameNumber, 3, 'frame number');

    ann.mathShape = new Rectangle(new Point2D(0, 0), new Point2D(2, 2));
    // move to the first volume: the centroid stays on the third
    position = geometry.indexToWorld(new Index([0, 0, 0, 0]));
    const centroid = ann.getCentroid();
    assert.deepEqual(
      centroid.getValues(), [1, 1, 0, 1000], 'centroid with volume id');
  });

  /**

   * Tests that {@link Annotation#setViewController} replaces legacy
   * plane cosines.
   *
   * @function module:tests/image~annotationSetVcLegacyCosines
   */
  test('Annotation setViewController replaces legacy plane cosines',
    () => {
      const ann = new Annotation();
      ann.referencedSopInstanceUID = 'sop-001';
      const origin = new Point3D(1, 2, 3);
      ann.planePoints = [
        origin,
        new Point3D(1, 0, 0),
        new Point3D(0, 0, 1)
      ];
      const vc = new MockViewController({
        uid: 'sop-001',

      assert.isTrue(ann.canView());
      assert.equal(ann.planePoints[0], origin, 'same origin');
      assert.isTrue(ann.planePoints[1].equals(new Point3D(1, 0, 0)),

describe('Annotation contains', () => {
  test('delegates to the math shape', () => {
    const annotation = new Annotation();
    annotation.mathShape = new Rectangle(new Point2D(0, 0), new Point2D(4, 4));
    assert.ok(annotation.contains(new Point2D(2, 2)), 'inside');
    assert.notOk(annotation.contains(new Point2D(6, 2)), 'outside');
    assert.ok(annotation.contains(new Point2D(6, 2), 2), 'within tolerance');
  });

  test('arrow uses the tip to tail segment', () => {
    const annotation = new Annotation();
    annotation.mathShape = new Point2D(0, 0);
    annotation.referencePoints = [new Point2D(10, 0)];
    assert.ok(annotation.contains(new Point2D(5, 1), 1), 'near the shaft');
    assert.notOk(annotation.contains(new Point2D(12, 0), 1), 'past the tail');
  });

  test('false without a math shape contains', () => {
    const annotation = new Annotation();
    assert.notOk(annotation.contains(new Point2D(0, 0)), 'no math shape');
    annotation.mathShape = new Path();
    assert.notOk(annotation.contains(new Point2D(0, 0)), 'path');
  });
});
